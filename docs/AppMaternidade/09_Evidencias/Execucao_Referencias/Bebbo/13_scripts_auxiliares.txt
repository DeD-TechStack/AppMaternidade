echo "jest exit=$? fim $(ts)" >> "$OUT/jest.meta"

npm audit --json > "$OUT/audit.json" 2> "$OUT/audit.err"

npm audit --omit=dev --json > "$OUT/audit_prod.json" 2> "$OUT/audit_prod.err"

echo "== outdated inicio $(ts)" > "$OUT/outdated.meta"

NODE_OPTIONS=--max-old-space-size=8192 ./node_modules/.bin/eslint app App.tsx --ext .js,.jsx,.ts,.tsx --ignore-pattern 'app/instances/*/assets/offlinecontent/**' --ignore-pattern 'app/instances/*/assets/locale/**' -f json -o "$OUT/eslint.json" > "$OUT/eslint.stdout" 2>&1
echo "eslint exit=$? fim $(ts)" >> "$OUT/eslint.meta"
echo "TUDO FEITO $(ts)" > "$OUT/DONE"

=================== summarize_checks.py
# Resume saídas de npm audit / npm outdated / eslint do clone do Bebbo.
import json, os, sys, collections
OUT = sys.argv[1]

def load(name):
    p = os.path.join(OUT, name)
    if not os.path.exists(p) or os.path.getsize(p) == 0:
        return None
    try:
        return json.load(open(p, encoding='utf-8'))
    except Exception as e:
        print(name, 'erro de parse:', e)
        return None

for name in ('audit.json', 'audit_prod.json'):
    a = load(name)
    if not a:
        print(name, ': vazio')
        continue
    meta = a.get('metadata', {})
    print('==', name, 'vulnerabilities:', meta.get('vulnerabilities'), 'deps:', meta.get('dependencies'))
    vulns = a.get('vulnerabilities', {})
    rows = []
    for pkg, v in vulns.items():
        if v.get('severity') in ('critical', 'high'):
            via = [x if isinstance(x, str) else x.get('title', '')[:70] for x in v.get('via', [])][:2]
            fix = v.get('fixAvailable')
            if isinstance(fix, dict):
                fix = '%s@%s major=%s' % (fix.get('name'), fix.get('version'), fix.get('isSemVerMajor'))
            rows.append((v.get('severity'), pkg, v.get('isDirect'), via, fix))
    for r in sorted(rows):
        print('  ', r)

o = load('outdated.json')
if o:
    major = []
    for pkg, v in o.items():
        if isinstance(v, list):
            v = v[0]
        cur = v.get('current') or v.get('wanted')
        lat = v.get('latest')
        try:
            cm = int(str(cur).split('.')[0]); lm = int(str(lat).split('.')[0])
        except Exception:
            cm = lm = None
        if cm is not None and lm is not None and lm > cm:
            major.append((pkg, cur, lat))
        elif pkg in ('react-native', 'realm', 'react', 'typescript'):
            major.append((pkg, cur, lat))
    print('== outdated: total=%d; com major atrás=%d' % (len(o), len(major)))
    for m in sorted(major):
        print('  ', m)

e = load('eslint.json')
if e:
    files = len(e)
    errs = sum(f.get('errorCount', 0) for f in e)
    warns = sum(f.get('warningCount', 0) for f in e)
    rules = collections.Counter()
    for f in e:
        for m in f.get('messages', []):
            rules[m.get('ruleId')] += 1
    print('== eslint: arquivos=%d erros=%d avisos=%d' % (files, errs, warns))
    for r, c in rules.most_common(12):
        print('  ', c, r)

# Extrai estrutura (sem copiar textos editoriais longos) das taxonomias e do conteudo offline do Bebbo.
# Uso: python -I inspect_taxonomy.py <pasta offlinecontent> <pasta locale>
import json, re, sys, os, collections

off = sys.argv[1]
loc = sys.argv[2]

def load_ts_object(path, varname):
    txt = open(path, encoding='utf-8').read()
    i = txt.index('=', txt.index(varname)) + 1
    body = txt[i:].strip()
    if body.endswith(';'):
        body = body[:-1]
    # remove trailing commas antes de } ou ]
    body = re.sub(r',(\s*[}\]])', r'\1', body)
    return json.loads(body)

tax = load_ts_object(os.path.join(off, 'taxonomies.ts'), 'taxonomydata')
print('idiomas na taxonomia offline:', len(tax), sorted(tax.keys()))
en = tax['en'][0]['allData']
print('chaves de taxonomia (en):', sorted(en.keys()))
ca = en.get('child_age', [])
print('child_age (en): %d faixas' % len(ca))
for c in ca:
    print('  id=%s name=%r days_from=%s days_to=%s buffers_days=%s age_bracket=%s' % (
        c.get('id'), c.get('name'), c.get('days_from'), c.get('days_to'), c.get('buffers_days'), c.get('age_bracket')))
print('campos de child_age:', sorted(ca[0].keys()) if ca else None)

for fname, var in [('article.ts', 'articledata'), ('ActivitiesData.ts', 'ActivitiesData'),
                   ('MileStonesData.ts', 'MileStonesData'), ('VideoArticleData.ts', 'VideoArticleData'),
                   ('FaqsData.ts', 'FaqsData')]:
    try:
        d = load_ts_object(os.path.join(off, fname), var)
    except Exception as e:
        print(fname, 'ERRO parse', e)
        continue
    counts = {k: len(v) for k, v in d.items()}
    print('%s: idiomas=%d contagem por idioma=%s' % (fname, len(d), counts))
    sample = d.get('en') or next(iter(d.values()))
    if sample:
        print('   campos do 1o item:', sorted(sample[0].keys()))
        lic = collections.Counter(str(x.get('licensed')) for x in sample)
        man = collections.Counter(str(x.get('mandatory')) for x in sample)
        print('   licensed=%s mandatory=%s' % (dict(lic), dict(man)))
        ua = sorted({str(x.get('updated_at'))[:7] for x in sample})
        print('   updated_at (ano-mes, min..max):', ua[0] if ua else None, '..', ua[-1] if ua else None)

# contagem de chaves de UI por locale
print('== strings de UI por locale ==')
for f in sorted(os.listdir(loc)):
    if not f.endswith('.ts') or f == 'availableLanguage.ts':
        continue
    txt = open(os.path.join(loc, f), encoding='utf-8').read()
    keys = re.findall(r'^\s*"?([A-Za-z0-9_]+)"?\s*:', txt, flags=re.M)
    print('  %-8s %5d chaves' % (f[:-3], len(set(keys))))

#!/usr/bin/env bash
# Coleta metadados públicos via gh CLI (somente leitura) para a ficha do Bebbo.
cd /c/Projetos/AppMaternidade/_referencias/bebbo || exit 1
R=UNICEFECAR/parenting-app-bebbo-mobile
C=UNICEFECAR/parenting-app-bebbo-CMS
run() { echo "\$ $*"; "$@" 2>&1; }
echo "# Metadados via API do GitHub (gh CLI, leitura pública) — $(date '+%Y-%m-%d %H:%M:%S %z')"
run gh api repos/$R --jq '{default_branch,archived,pushed_at,updated_at,created_at,license:.license.spdx_id,stars:.stargazers_count,forks:.forks_count,open_issues:.open_issues_count,size_kb:.size}'
run git ls-remote origin HEAD refs/heads/main
run gh api repos/$R/branches --paginate --jq '.[]|.name'
echo "\$ gh pr list -R $R --state open (número, data, autor, título)"
gh pr list -R $R --state open --limit 30 --json number,title,createdAt,author --jq '.[]|"#\(.number) \(.createdAt[0:10]) \(.author.login) | \(.title)"' 2>&1
echo "\$ gh issue list -R $R --state open (número, data, título)"
gh issue list -R $R --state open --limit 30 --json number,title,createdAt --jq '.[]|"#\(.number) \(.createdAt[0:10]) | \(.title)"' 2>&1
run git for-each-ref refs/tags/Prod_v2.5.0 refs/tags/Prod_v2.4.4 --format='%(refname:short) %(creatordate:short)'
echo
echo "# Repositório do CMS (backend de conteúdo)"
run gh api repos/$C --jq '{default_branch,archived,pushed_at,license:.license.spdx_id,language,open_issues:.open_issues_count}'
run gh api repos/$C/commits/main --jq '.sha + " " + .commit.committer.date'
echo
echo "# Bibliotecas usadas pelo app (situação upstream)"
for r in realm/realm-js zo0r/react-native-push-notification itinance/react-native-fs DylanVann/react-native-fast-image; do
  run gh api repos/$r --jq '{full_name,archived,pushed_at}'
done
echo "\$ (README do realm-js, primeiras linhas)"
gh api -H "Accept: application/vnd.github.raw" repos/realm/realm-js/readme 2>&1 | head -6
echo "\$ versões instaladas em node_modules (após npm ci --ignore-scripts)"
for p in react-native react realm @react-native-firebase/app react-native-push-notification redux-persist typescript i18next; do
  printf '%s ' "$p"; node -e "console.log(require('./node_modules/$p/package.json').version)" 2>&1
done
echo "fim: $(date '+%Y-%m-%d %H:%M:%S %z')"

=================== treesize.py
import json, sys, collections
d = json.load(open(sys.argv[1], encoding='utf-8'))
by2 = collections.Counter(); cnt2 = collections.Counter()
ext = collections.Counter()
big = []
for t in d['tree']:
    if t['type'] != 'blob': continue
    p = t['path']; s = t.get('size', 0)
    parts = p.split('/')
    k = '/'.join(parts[:2]) if len(parts) > 1 else parts[0]
    by2[k] += s; cnt2[k] += 1
    e = p.rsplit('.', 1)[-1].lower() if '.' in parts[-1] else '(none)'
    ext[e] += s
    big.append((s, p))
print("== top dirs (2 levels) by size ==")
for k, v in by2.most_common(30):
    print(f"{v/1e6:9.2f} MB  {cnt2[k]:5d} files  {k}")
print("== by extension ==")
for k, v in ext.most_common(20):
    print(f"{v/1e6:9.2f} MB  {k}")
print("== biggest files ==")
for s, p in sorted(big, reverse=True)[:25]:
    print(f"{s/1e6:9.2f} MB  {p}")
